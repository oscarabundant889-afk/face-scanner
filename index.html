/* ========================================================= FACE SELECTION FINAL AUTO READY AFTER REMARK G / AK IS FILLED ========================================================= */ const FACE_CONTROL_SHEET = "_FACE_CONTROL"; const FACE_TARGET_COLUMNS = [ 1, // A 5, // E 20, // T 24, // X 31, // AE 35 // AI ]; /* ========================================================= SELECTION CHANGE ========================================================= */ function onSelectionChange(e) { try { if (!e || !e.range) return; const range = e.range; const sheet = range.getSheet(); const sheetName = sheet.getName().trim(); /* * Hanya attendance sheet */ if ( !/^\d{2}-\d{2}(MS|NS)$/i.test( sheetName ) ) { return; } const row = range.getRow(); const column = range.getColumn(); if (row < 5) { return; } if ( FACE_TARGET_COLUMNS.indexOf( column ) === -1 ) { return; } updateFaceTargetFromSelection( e.source, sheet, row, column ); } catch (error) { console.error( "FACE SELECTION ERROR:", error ); } } /* ========================================================= AUTO CHECK WHEN G / AK IS EDITED ========================================================= */ function onEdit(e) { try { if (!e || !e.range) { return; } const range = e.range; const sheet = range.getSheet(); const sheetName = sheet.getName().trim(); /* * Hanya attendance sheet */ if ( !/^\d{2}-\d{2}(MS|NS)$/i.test( sheetName ) ) { return; } const row = range.getRow(); const column = range.getColumn(); if (row < 5) { return; } /* * G = Remark Group 1 * AK = Remark Group 3 */ if ( column !== 7 && column !== 37 ) { return; } const ss = e.source; let control = ss.getSheetByName( FACE_CONTROL_SHEET ); if (!control) { control = ss.insertSheet( FACE_CONTROL_SHEET ); control.hideSheet(); } const data = control .getRange("A1:E1") .getValues()[0]; const targetSheet = String( data[0] || "" ).trim(); const targetRow = Number( data[1] || 0 ); const targetColumn = Number( data[2] || 0 ); /* * Kalau sedang menunggu E * dan G baru diisi */ if ( column === 7 && targetSheet === sheetName && targetRow === row && targetColumn === 5 ) { updateFaceTargetFromSelection( ss, sheet, row, 5 ); return; } /* * Kalau sedang menunggu AI * dan AK baru diisi */ if ( column === 37 && targetSheet === sheetName && targetRow === row && targetColumn === 35 ) { updateFaceTargetFromSelection( ss, sheet, row, 35 ); return; } } catch (error) { console.error( "FACE REMARK EDIT ERROR:", error ); } } /* ========================================================= UPDATE TARGET ========================================================= */ function updateFaceTargetFromSelection( ss, sheet, row, column ) { let control = ss.getSheetByName( FACE_CONTROL_SHEET ); if (!control) { control = ss.insertSheet( FACE_CONTROL_SHEET ); control.hideSheet(); } /* * E membutuhkan A * X membutuhkan T * AI membutuhkan AE */ const prerequisiteMap = { 5: 1, 24: 20, 35: 31 }; const requiredColumn = prerequisiteMap[column]; if (requiredColumn) { const employee = String( sheet .getRange( row, requiredColumn ) .getDisplayValue() ).trim(); if (!employee) { setFaceControl( control, sheet.getName(), row, column, "WAITING" ); updateFaceStatusSheet( sheet, row, column, "EMPLOYEE REQUIRED" ); SpreadsheetApp.flush(); return; } } /* * G wajib untuk E */ if (column === 5) { const remark = String( sheet .getRange(row, 7) .getDisplayValue() ).trim(); if (!remark) { setFaceControl( control, sheet.getName(), row, column, "WAITING" ); updateFaceStatusSheet( sheet, row, column, "REMARK REQUIRED" ); SpreadsheetApp.flush(); return; } } /* * AK wajib untuk AI */ if (column === 35) { const remark = String( sheet .getRange(row, 37) .getDisplayValue() ).trim(); if (!remark) { setFaceControl( control, sheet.getName(), row, column, "WAITING" ); updateFaceStatusSheet( sheet, row, column, "REMARK REQUIRED" ); SpreadsheetApp.flush(); return; } } /* * Target sudah terisi */ const currentValue = String( sheet .getRange( row, column ) .getDisplayValue() ).trim(); if (currentValue !== "") { setFaceControl( control, sheet.getName(), row, column, "FILLED" ); updateFaceStatusSheet( sheet, row, column, "ALREADY COMPLETED" ); SpreadsheetApp.flush(); return; } /* * READY */ setFaceControl( control, sheet.getName(), row, column, "READY" ); updateFaceStatusSheet( sheet, row, column, "READY FOR FACE SCAN" ); SpreadsheetApp.flush(); } /* ========================================================= WRITE CONTROL ========================================================= */ function setFaceControl( control, sheetName, row, column, status ) { control .getRange("A1:E1") .setValues([[ sheetName, row, column, new Date(), status ]]); } /* ========================================================= STATUS PANEL AL:AM ========================================================= */ function updateFaceStatusSheet( sheet, row, column, message ) { const cfg = { 1: { employee: 1, name: 2, timeout: 3, timeoutStamp: 4, timeIn: 5, timeInStamp: 6, remark: 7 }, 5: { employee: 1, name: 2, timeout: 3, timeoutStamp: 4, timeIn: 5, timeInStamp: 6, remark: 7 }, 20: { employee: 20, name: 21, timeout: 22, timeoutStamp: 23, timeIn: 24, timeInStamp: 25, remark: null }, 24: { employee: 20, name: 21, timeout: 22, timeoutStamp: 23, timeIn: 24, timeInStamp: 25, remark: null }, 31: { employee: 31, name: 32, timeout: 33, timeoutStamp: 34, timeIn: 35, timeInStamp: 36, remark: 37 }, 35: { employee: 31, name: 32, timeout: 33, timeoutStamp: 34, timeIn: 35, timeInStamp: 36, remark: 37 } }; const c = cfg[column]; if (!c) { return; } const get = function(col) { if (!col) { return ""; } return String( sheet .getRange( row, col ) .getDisplayValue() || "" ).trim(); }; const values = [ ["TARGET", columnToLetterFace(column) + row], ["STATUS", message], ["EMPLOYEE NO", get(c.employee) || "NOT YET"], ["NAME", get(c.name) || "NOT YET"], ["TIME OUT", get(c.timeout) || "NOT YET"], ["SYSTEM TIME OUT", get(c.timeoutStamp) || "NOT YET"], ["TIME IN", get(c.timeIn) || "NOT YET"], ["SYSTEM TIME IN", get(c.timeInStamp) || "NOT YET"], ["REMARK", get(c.remark) || "NOT YET"] ]; sheet .getRange("AL1:AM12") .breakApart() .clearContent(); sheet .getRange("AL1:AM1") .merge() .setValue( "FACE ATTENDANCE STATUS" ); sheet .getRange( "AL2:AM10" ) .setValues( values ); sheet .getRange( "AL1:AM10" ) .setVerticalAlignment( "middle" ) .setBorder( true, true, true, true, true, true ); sheet .getRange("AL1:AM1") .setFontWeight("bold") .setHorizontalAlignment( "center" ) .setFontSize(14); sheet .getRange("AL2:AL10") .setFontWeight("bold"); sheet.setColumnWidth( 38, 170 ); sheet.setColumnWidth( 39, 250 ); SpreadsheetApp.flush(); } /* ========================================================= COLUMN NUMBER → LETTER ========================================================= */ function columnToLetterFace( column ) { let letter = ""; while (column > 0) { const temp = (column - 1) % 26; letter = String.fromCharCode( temp + 65 ) + letter; column = Math.floor( (column - temp - 1) / 26 ); } return letter; } /* ========================================================= TEST ========================================================= */ function TEST_FACE_SELECTION() { const ss = SpreadsheetApp .getActiveSpreadsheet(); let control = ss.getSheetByName( FACE_CONTROL_SHEET ); if (!control) { control = ss.insertSheet( FACE_CONTROL_SHEET ); control.hideSheet(); } control .getRange("A1:E1") .setValues([[ "TEST", 999, 1, new Date(), "READY" ]]);
